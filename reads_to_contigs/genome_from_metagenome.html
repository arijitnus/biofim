## From raw reads to assembly 
#%% 
# Removing sequencing adapters using cutadapt
# more info at https://cutadapt.readthedocs.io/en/stable/guide.html
# Reference: DOI: https://doi.org/10.14806/ej.17.1.200
# detail: https://cutadapt.readthedocs.io/en/stable/
# Intallation using conda
conda create -n cutadaptenv cutadaptenv
conda activate cutadaptenv
cutadapt --version # for checking
# Using cutadapt
cutadapt -a AGATCGGAAGAGCACACGTCTGAACTCCAGTCA -A AGATCGGAAGAGCGTCGTGTAGGGAAAGAGTGT -o outread1.fastq -p outread2.fastq \ 
reads_1.fastq reads_2.fastq --error-rate 0.2 --minimum-length 75 --no-indels
#%% 
# Trimming low quality regions using BBduk (part of BBmap software package)
# link : https://sourceforge.net/projects/bbmap/
# info at : http://seqanswers.com/forums/showthread.php?t=42776
bbduk.sh -Xmx1g in=outread_1.fastq in2=outread_2.fastq out1=clean_1.fq out2=clean_2.fq trimq=20 qtrim=rl minlen=75

#5-10 min for 15 Gb data
#%% 
# Metagenome Comparison
# To compare similarity betweent the metagenomes by MASH
# https://github.com/lskatz/mashtree
conda install -c bioconda mashtree

mashtree_bootstrap.pl --reps 100 --numcpus 20 *.fq.gz -- --min-depth 0 > mashtree.bootstrap.dnd
#%% 
# Assembly using Megahit
# paper: Sczyrba A, Hofmann P, Belmann P, Koslicki D, Janssen S, Dröge J, Gregor I, Majda S, Fiedler J, Dahms E, Bremges A. Critical assessment of metagenome interpretation—a benchmark of metagenomics software. Nature methods. 2017 Nov;14(11):1063-71.
# installation: conda install -c bioconda megahit
megahit -1 clean_1.fq -2 clean_2.fq --k-min 27 --k-max 197 --k-step 10 –min_contigs_len 1000 -t 10

#the detail about the assembled contigs will be at the end of megahit log file
# time : 1hrs for nearly 15Gb data with 24 thread
#%% 
# For co-assembly, interleaving of reads are need to be done
# script link: https://github.com/dariober/bioinformatics-cafe/tree/master/interleaveFastq
interleave_fastq.sh forward.fastq reverse.fastq > meta1.fastq
# Then provides this interleaved reads during assembly
megahit --12 meta1.fastq, meta2.fastq, meta3.fastq --k-min 27 --k-max 197 --k-step 10 –min_contigs_len 1000 -t 10

(check megahit githib for usage of --k-min max and other parameter depending upon metagenome source and sequencing depth)
#%% 
# Read containment was estimated by mapping the trimmed paired-end reads used for each assembly to the assembled contigs 
#%% 
# bowtie2
# installation: conda install bowtie2
# detail: http://bowtie-bio.sourceforge.net/bowtie2/index.shtml
# for co-assembly either one can go for separate mapping or combined mapping
#%% 
# Indexing
bowtie2-build contigs.fa contigs_indexed
#%% 
# Mapping
    bowtie2 –fr -x contigs_indexed -1 read1.fq -2 read2.fq --no-unal -X 1000 -S contigs.sam -p 10
#%% 
# sam to bam (more detail on options http://www.htslib.org/doc/samtools-view.html)
samtools view -bS contigs.sam > contigs.bam --threads 10
#%% 
# Sorting
samtools sort contigs.bam > contigs_sort.bam --threads 10
#%% 
# all in one line
bowtie2 –fr -x contigs_indexed -p 20 -1 read1.fq -2 read2.fq --no-unal -X 1000 | samtools view -Shu - | samtools sort - -o sort.bam

#8 hrs for 10metagenome
#%% md 
## Genome Binning and bin purification 
#%% 
# reading: Meyer F, et al. AMBER: assessment of metagenome binners. GigaScience. 2018 Jun;7(6):giy069.
#%% 
# 3 tools for binning: MetaBat2, Concoct, MaxBin2 (based on sequence composition and differential coverage)
# installation
# metabat2 detail at : https://bitbucket.org/berkeleylab/metabat/src/master/
# concoct installation: conda create -n concoct_env python=3 concoct
# maxbin installation: conda install -c bioconda maxbin2
#%% 
# Using metabat2
# Generating depth file (jgi_summarize_bam_contig_depths file comes along with metabat2 program)
jgi_summarize_bam_contig_depths contigs_sort.bam depth.txt
metabat2 -I contigs.fa -a depth.txt --minS 80 -o out_bins
#%% 
# Using maxbin2
# maxbin2 details for creating abundance file: https://sourceforge.net/projects/maxbin2/files/
# abundance file can be createed from metabat2 depth file by simple editing (cut -f1,3 and sed -i '1d')
run_MaxBin.pl -min_contig_length 2500 -contig contigs.fa -abund myabund -out myout -thread 10
#%% 
# Using concoct
#1 cut contigs into smaller parts:
cut_up_fasta.py contigs.fa -c 10000 -o 0 --merge_last -b contigs_10K.bed > contigs_10K.fa

#2 Generate table with coverage depth information per sample and subcontig. 
concoct_coverage_table.py contigs_10K.bed mapping/Sample*.sorted.bam > coverage_table.tsv

#3 running concoct
concoct --composition_file contigs_10K.fa -l 2500 --coverage_file coverage_table.tsv -b concoct_output/

#4 Merge subcontig clustering into original contig clustering:
merge_cutup_clustering.py concoct_output/clustering_gt1000.csv > concoct_output/clustering_merged.csv

#5 Extract bins as individual FASTA
mkdir concoct_output/fasta_bins
extract_fasta_bins.py contigs.fa concoct_output/clustering_merged.csv --output_path concoct_output/fasta_bins
#%% 
# Aggregating and dereplicating all the bins using DAS Tool
# paper: doi: 10.1038/s41564-018-0171-1
# installation: conda install -c bioconda das_tool
# detail: https://github.com/cmks/DAS_Tool

# preparing tsv file for dastool
# from maxbin and metabat bins
bash ~/das_tool-1.1.1-2/src/Fasta_to_Scaffolds2Bin.sh -i ~/bin_folder -e fa > maxbin.scaffolds2bin.tsv

# from concoct coverage csv file
perl -pe "s/,/\tconcoct./g;" concoct_clustering_gt1000.csv > concoct.scaffolds2bin.tsv

#running das tool
DAS_Tool  -i ~/concoct.scaffolds2bin.tsv,~/maxbin.scaffolds2bin.tsv,~/metabat.scaffolds2bin.tsv \ 
-l concoct,maxbin,metabat -c ~/contigs.fa -o das_output/DASbin -t 10 --search_engine diamond \ 
--write_bins 1 --score_threshold 0
#%% 
# Bin refinement using RefineM
#1. Generating index bam file from sorted bam file
samtools index scaffold_sort.bam

#2. To identify scaffolds with genomic properties that are divergent from the expect values for a bin
refinem scaffold_stats -c 10 scaffold.fa ~/das_output/DASbin scaffold_sort.bam --genome_ext fa output-folder

#3. Scaffolds with divergent genomic properties can then be identified
refinem outliers output-folder/scaffold_stats.tsv lieroutput_dir

#4. Contaminating scaffolds can be removed from bins
refinem filter_bins path-to-bins lieroutput_dir/outliers.tsv filtered_output --genome_ext fa
#%% 
# MAGpurify v1.0 to identify and remove taxonomically discordant contigs
# detail: https://github.com/snayfach/MAGpurify
# reference: https://doi.org/10.1038/s41586-019-1058-x
# installation: conda create --name magpurify python=2.7
# conda install -c conda-forge -c bioconda magpurify
# download database: wget https://zenodo.org/record/3688811/files/MAGpurify-db-v1.0.tar.bz2?download=1
# rename and untar it
# add to path: export MAGPURIFYDB=/path/to/MAGpurify-db-v1.0
# module can be phylo-markers, clade-markers.
magpurify <module> <input mag> <output directory>
magpurify clean-bin <input mag> <output directory> <output mag>
#%% 
# Species and subspecies level MAGs using dRep
# reference: https://doi.org/10.1038/ismej.2017.126
# detail: https://github.com/MrOlm/drep
# installation: pip install drep
dRep dereplicate -comp 50 -con 10 -sani 0.95/0.99 S_algorithm gANI outout_directory -g path/to/genomes/*.fasta
